site-nosso/
│
├── backend/                  # O motor do site (Node.js / Express)
│   ├── package.json          # Criado com 'npm init'
│   ├── .env                  # Onde vais pôr as passwords da base de dados e chaves secretas (nunca partilhar!)
│   └── src/
│       ├── server.js         # O ficheiro principal que arranca o servidor
│       ├── config/           # Ligações à base de dados (ex: mongo.js ou db.js)
│       ├── models/           # Os moldes dos teus dados (ex: User.js, Message.js, Photo.js)
│       ├── routes/           # Os caminhos (ex: /api/login, /api/mensagens)
│       ├── controllers/      # A lógica (o que acontece quando visitas a rota)
│       └── middlewares/      # Segurança (ex: verificar se o utilizador fez login antes de ver as fotos)
│
└── frontend/                 # O aspeto visual do site (React ou Next.js)
    ├── package.json          # Criado com 'npx create-react-app' ou similar
    ├── public/               # Ícones e ficheiros públicos
    └── src/
        ├── App.js            # O componente principal onde configuras as rotas visuais
        ├── components/       # Peças reutilizáveis (ex: MenuPrincipal, Botao, CartaoMemoria)
        ├── pages/            # As abas que pediste
        │   ├── Login.jsx
        │   ├── Dashboard.jsx # O menu inicial
        │   ├── Mensagens.jsx
        │   ├── Textos.jsx
        │   ├── Fotos.jsx
        │   └── Memorias.jsx
        ├── services/         # Funções para falar com o backend (ex: api.js)
        └── styles/           # O teu CSS / Estilos
